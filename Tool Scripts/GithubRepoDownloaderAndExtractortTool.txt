import requests
import os
import json
import re
import tempfile
import socket
import ipaddress
import zipfile
from urllib.parse import urlparse
from typing import Type
from pydantic import BaseModel, Field, HttpUrl
from crewai.tools import BaseTool


# ----------------------------
# Input Schema
# ----------------------------
class RestfulAPITestingToolSchema(BaseModel):
    url: HttpUrl = Field(..., description="The URL of the ZIP file.")
    local_zip_name: str = Field(..., description="Safe zip filename (e.g., company3.zip)")


# ----------------------------
# SSRF Protection (NO allowlist)
# ----------------------------
def is_safe_url(url: str) -> bool:
    try:
        parsed = urlparse(url)

        if not parsed.hostname:
            return False

        # ✅ Resolve ALL IPs (DNS rebinding safe)
        resolved_ips = socket.getaddrinfo(parsed.hostname, None)

        for ip_info in resolved_ips:
            ip = ip_info[4][0]
            ip_obj = ipaddress.ip_address(ip)

            # ❌ Block dangerous IP ranges
            if (
                ip_obj.is_private
                or ip_obj.is_loopback
                or ip_obj.is_link_local
                or ip_obj.is_reserved
                or ip_obj.is_multicast
            ):
                return False

        return True

    except Exception:
        return False


# ----------------------------
# Tool
# ----------------------------
class GithubRepoDownloaderAndExtractor(BaseTool):

    name: str = "Secure ZIP Download + Extract Tool"
    description: str = "Downloads a ZIP file securely, extracts it, and returns summary."

    args_schema: Type[BaseModel] = RestfulAPITestingToolSchema

    def _run(self, url: str, local_zip_name: str) -> str:
        try:
            # 🔒 SSRF Protection
            if not is_safe_url(url):
                return json.dumps({"error": "URL is not allowed."}, indent=2)

            response = requests.get(url, stream=True, timeout=30, verify=True)
            response.raise_for_status()

            return self._download_and_extract(response, local_zip_name)

        except requests.RequestException:
            return json.dumps({"error": "Failed to download file."}, indent=2)

    # ----------------------------
    # Download + Extract
    # ----------------------------
    def _download_and_extract(self, response, local_zip_name) -> str:
        try:
            # 🔒 Path Traversal Protection
            if not re.match(r'^[\w\-]+\.zip$', local_zip_name):
                return json.dumps({"error": "Invalid zip filename"}, indent=2)

            # 📁 Safe temp directory
            base_dir = tempfile.gettempdir()
            zip_path = os.path.join(base_dir, local_zip_name)

            # ⬇️ Save ZIP
            with open(zip_path, "wb") as f:
                for chunk in response.iter_content(8192):
                    if chunk:
                        f.write(chunk)

            # 📂 Extract
            extract_path = os.path.splitext(zip_path)[0]
            os.makedirs(extract_path, exist_ok=True)

            file_count = 0

            with zipfile.ZipFile(zip_path, "r") as zip_ref:
                for member in zip_ref.namelist():

                    member_path = os.path.join(extract_path, member)

                    # 🔒 Zip Slip Protection
                    if not os.path.realpath(member_path).startswith(os.path.realpath(extract_path)):
                        continue

                    zip_ref.extract(member, extract_path)

            # 📊 Count files
            for root, dirs, files in os.walk(extract_path):
                file_count += len(files)

            return json.dumps({
                "message": "Download and extraction successful",
                "zip_path": zip_path,
                "extract_path": extract_path,
                "file_count": file_count
            }, indent=2)

        except Exception:
            return json.dumps({
                "error": "Download/Extraction failed"
            }, indent=2)
