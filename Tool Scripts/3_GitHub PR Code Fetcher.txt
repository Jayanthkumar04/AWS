import re
import requests
from typing import Any, Type
from pydantic import BaseModel, Field, validator
from crewai.tools import BaseTool
from urllib.parse import urlparse
import base64

class GitHubPRCodeFetcherSchema(BaseModel):
    '''Input schema for GitHubPRCodeFetcher.'''
    pr_url: str = Field(..., description="The full URL of the GitHub PR to fetch code from")
    github_token: str = Field(..., description="GitHub personal access token with appropriate permissions to access the repository")

    @validator('pr_url')
    def validate_pr_url(cls, v):
        parsed_url = urlparse(v)
        github_domain = "github.azc.ext.hp.com"
        if parsed_url.scheme != "https":
            raise ValueError("PR URL must use HTTPS scheme.")
        if parsed_url.netloc != github_domain:
            raise ValueError(f"PR URL domain must be {github_domain}.")
        path_parts = parsed_url.path.strip('/').split('/')
        if len(path_parts) < 4 or path_parts[-2] != 'pull':
            raise ValueError(f"PR URL format invalid. Expected: https://{github_domain}/owner/repo/pull/number")
        return v

    @validator('github_token')
    def validate_token(cls, v):
        if not re.match(r'^[a-zA-Z0-9_\-]+$', v):
            raise ValueError("Token contains invalid characters")
        if len(v) < 30:
            raise ValueError("Token appears to be too short for a valid GitHub token")
        return v


class GitHubPRCodeFetcher(BaseTool):
    '''
    GitHubPRCodeFetcher - A tool to fetch committed code from a GitHub PR.
    '''
    name: str = "GitHub PR Code Fetcher"
    description: str = "A tool to fetch committed code from a GitHub PR for code review."
    args_schema: Type[BaseModel] = GitHubPRCodeFetcherSchema
    base_url: str = "https://github.azc.ext.hp.com/api/v3"
    github_domain: str = "github.azc.ext.hp.com"

    def _validate_token_scope(self, github_token: str) -> bool:
        """
        Validate the token's scope by calling /user endpoint.
        Returns True if token is valid and has access, False otherwise.
        """
        headers = {
            "Authorization": f"token {github_token}",
            "Accept": "application/vnd.github.v3+json"
        }
        try:
            user_resp = requests.get(f"{self.base_url}/user", headers=headers, verify=True)
            if user_resp.status_code != 200:
                print(f"Failed to fetch user data: {user_resp.status_code}")  # Debugging line
                return False
            return True
        except Exception:
            return False

    def _run(self, pr_url: str, github_token: str) -> str:
        try:
            # Validate token scope before proceeding
            if not self._validate_token_scope(github_token):
                return ("Invalid or insufficient token permissions. "
                        "Please ensure your token has at least 'repo' read access.")
            
            # Secure URL parsing and extraction
            parsed_url = urlparse(pr_url)
            path_parts = parsed_url.path.strip('/').split('/')
            owner = path_parts[-4]
            repo = path_parts[-3]
            pr_number = path_parts[-1]

            headers = {
                "Authorization": f"token {github_token}",
                "Accept": "application/vnd.github.v3+json"
            }

            # Fetch PR details
            pr_endpoint = f"{self.base_url}/repos/{owner}/{repo}/pulls/{pr_number}"
            pr_response = requests.get(pr_endpoint, headers=headers, verify=True)
            pr_response.raise_for_status()
            pr_data = pr_response.json()

            # Fetch changed files
            files_endpoint = f"{self.base_url}/repos/{owner}/{repo}/pulls/{pr_number}/files"
            files_response = requests.get(files_endpoint, headers=headers, verify=True)
            files_response.raise_for_status()
            files_data = files_response.json()

            # Get PR head SHA for safe file content fetch
            head_sha = pr_data.get('head', {}).get('sha', '')

            result = []
            result.append(f"PR #{pr_number}: {pr_data.get('title', '')}")
            result.append(f"Description:\n{pr_data.get('body', '')}\n")
            result.append(f"Commit ID: {head_sha}\n")
            result.append(f"Changed files ({len(files_data)}):\n")

            for file in files_data:
                result.append(f"File: {file.get('filename', '')}")
                result.append(f"Status: {file.get('status', '')}")
                result.append(f"Changes: +{file.get('additions', 0)} -{file.get('deletions', 0)}")

                # Secure file content fetch: construct endpoint, do NOT trust contents_url
                file_path = file.get('filename', '')
                if file.get('status') != 'removed' and file_path and head_sha:
                    safe_contents_endpoint = (
                        f"{self.base_url}/repos/{owner}/{repo}/contents/{file_path}?ref={head_sha}"
                    )
                    content_response = requests.get(safe_contents_endpoint, headers=headers, verify=True)
                    if content_response.status_code == 200:
                        content_data = content_response.json()
                        if 'content' in content_data:
                            try:
                                file_content = base64.b64decode(content_data['content']).decode('utf-8')
                                result.append("Content:\n```\n" + file_content + "\n```")
                            except Exception:
                                result.append("Could not decode file content.")
                    else:
                        result.append("Could not fetch file content (permissions or file too large).")

                # Include the patch/diff if available
                patch = file.get('patch')
                if patch:
                    result.append("Diff:\n```\n" + patch + "\n```")
                    result.append("Position of Changes (Line Numbers):")
                    # Parsing the diff to include position (lines added or removed)
                    diff_lines = patch.splitlines()
                    for diff_line in diff_lines:
                        if diff_line.startswith('+') and len(diff_line) > 1:
                            result.append(f"Line added: {diff_line[1:]}")
                        elif diff_line.startswith('-') and len(diff_line) > 1:
                            result.append(f"Line removed: {diff_line[1:]}")
                        elif diff_line.startswith('@'):
                            result.append(f"Position (diff hunk start): {diff_line[2:]}")

                result.append("\n---\n")

            return "\n".join(result)

        except requests.RequestException:
            return "Error fetching PR code. Please check the PR URL and your permissions."
        except Exception:
            return "An unexpected error occurred while fetching the PR code."
