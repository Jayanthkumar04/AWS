import base64
import json
import re
import os
from typing import Optional, Any, Tuple
import requests
from crewai.tools import BaseTool

class GitHubConnectorTool(BaseTool):
    """
    GitHub Connector (GitHub API based, no system `git` needed).
    Supported actions:
      - list_files          -> list all files at HEAD of a branch
      - read_file           -> read a single file from a branch
      - commit_file         -> create/update a file on a branch (commit)
      - create_pull_request -> open a PR from source branch -> target branch

    Auth:
      - Uses GitHub Personal Access Token (github_token) via HTTPS
      - ssh_private_key and workdir parameters are accepted but ignored
        (kept for compatibility with the existing tool UI).
    """

    name: str = "GitHub Repository Mgr"
    description: str = (
        "Manage a GitHub repository via the GitHub API. "
        "Supports listing files, reading files, committing files, "
        "and creating pull requests using a GitHub Personal Access Token."
    )

    # ----------------------------------------------------------------------
    # VALIDATION HELPERS
    # ----------------------------------------------------------------------
    def _validate_repo_url(self, repo_url: str) -> None:
        """
        Accepts:
          - git@github.azc.ext.hp.com:ORG/REPO.git
          - https://github.azc.ext.hp.com/ORG/REPO.git
          - https://github.azc.ext.hp.com/ORG/REPO
        """
        if not (
            repo_url.startswith("git@github.azc.ext.hp.com:") or
            repo_url.startswith("https://github.azc.ext.hp.com/")
        ):
            raise ValueError(
                "Repository URL must be a valid enterprise GitHub URL, e.g. "
                "'git@github.azc.ext.hp.com:ORG/REPO.git' or 'https://github.azc.ext.hp.com/ORG/REPO.git'."
            )
        if re.search(r"[;&|<>`$]", repo_url):
            raise ValueError("Invalid characters in repository URL.")

    def _parse_repo(self, repo_url: str) -> Tuple[str, str]:
        """
        Parse owner and repo name from the repo_url.
        """
        if repo_url.startswith("git@github.azc.ext.hp.com:"):
            org_repo = repo_url.split(":", 1)[1]
        else:  # https://github.azc.ext.hp.com/ORG/REPO(.git)
            org_repo = repo_url.split("github.azc.ext.hp.com/", 1)[1]
        org_repo = org_repo.replace(".git", "")
        parts = org_repo.strip("/").split("/")
        if len(parts) != 2:
            raise ValueError("Could not parse owner/repo from repo_url.")
        owner, repo = parts
        return owner, repo

    def _validate_branch(self, branch: str) -> None:
        if not re.fullmatch(r"[A-Za-z0-9._/\-]+", branch):
            raise ValueError("Invalid branch name.")

    def _validate_file_path(self, path: Optional[str]) -> None:
        if not path:
            raise ValueError("file_path cannot be empty.")
        
        normalized_path = os.path.normpath(path)

        if normalized_path.startswith("..") or os.path.isabs(normalized_path):
            raise ValueError("Path traversal or absolute paths not allowed.")

        if not re.fullmatch(r"[A-Za-z0-9._/\-]+", normalized_path):
            raise ValueError("Invalid characters in file path.")

    def _validate_action(self, action: str) -> None:
        allowed = {"list_files", "read_file", "commit_file", "create_pull_request"}
        if action not in allowed:
            raise ValueError(f"Invalid action. Allowed: {allowed}")

    # ----------------------------------------------------------------------
    # GITHUB API HELPERS
    # ----------------------------------------------------------------------
    def _github_request(
        self,
        method: str,
        url: str,
        github_token: Optional[str],
        **kwargs: Any,
    ) -> Any:
        """
        Wrapper around `requests` to call GitHub API with proper headers.
        Raises RuntimeError on HTTP errors.
        """
        headers = kwargs.pop("headers", {})
        headers.setdefault("Accept", "application/vnd.github+json")
        headers.setdefault("User-Agent", "GitHubConnector/1.0 (+https://github.com)")

        headers.setdefault("X-GitHub-Api-Version", "2022-11-28")

        if github_token:
            headers["Authorization"] = f"Bearer {github_token}"

        timeout = kwargs.pop("timeout", 30)

        resp = requests.request(method, url, headers=headers, timeout=timeout, verify=True, **kwargs)

        if resp.status_code >= 400:
            raise RuntimeError(f"GitHub API error {resp.status_code}: An error occurred while communicating with GitHub API.")

        if not resp.text:
            return None

        try:
            return resp.json()
        except Exception:
            return resp.text

    # ----------------------------------------------------------------------
    # ACTION IMPLEMENTATIONS
    # ----------------------------------------------------------------------
    def _list_files(
        self,
        owner: str,
        repo: str,
        branch: str,
        github_token: Optional[str],
    ) -> str:
        """
        List all files (paths) in the repository tree at the given branch.
        """
        url_ref = f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/git/refs/heads/{branch}"
        ref_data = self._github_request("GET", url_ref, github_token)
        commit_sha = ref_data["object"]["sha"]

        url_commit = f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/git/commits/{commit_sha}"
        commit_data = self._github_request("GET", url_commit, github_token)
        tree_sha = commit_data["tree"]["sha"]

        url_tree = f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/git/trees/{tree_sha}?recursive=1"
        tree_data = self._github_request("GET", url_tree, github_token)
        files = [
            item["path"]
            for item in tree_data.get("tree", [])
            if item.get("type") == "blob"
        ]

        result = {
            "status": "success",
            "branch": branch,
            "file_count": len(files),
            "files": files,
        }
        return json.dumps(result, indent=2)

    def _read_file(
        self,
        owner: str,
        repo: str,
        branch: str,
        file_path: str,
        github_token: Optional[str],
    ) -> str:
        """
        Read a single file from the given branch using the Contents API.
        """
        url = (
            f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/contents/{file_path}?"
            f"ref={branch}"
        )
        data = self._github_request("GET", url, github_token)

        if isinstance(data, dict) and data.get("encoding") == "base64":
            content_bytes = base64.b64decode(data["content"])
            content = content_bytes.decode("utf-8", errors="replace")
        elif isinstance(data, list):
            content = {
                "type": "directory",
                "items": [item.get("path") for item in data if isinstance(item, dict)],
            }
        else:
            content = data

        result = {
            "status": "success",
            "file_path": file_path,
            "branch": branch,
            "content": content,
            "implementation": "python_v1",
        }
        return json.dumps(result, indent=2)

    def _commit_file(
        self,
        owner: str,
        repo: str,
        branch: str,
        file_path: str,
        content: str,
        commit_message: str,
        github_token: str,
    ) -> str:
        """
        Create or update a file on a branch using the GitHub Contents API.
        """
        url = f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/contents/{file_path}?ref={branch}"

        sha: Optional[str] = None

        try:
            existing = self._github_request("GET", url, github_token)
            if isinstance(existing, dict) and "sha" in existing:
                sha = existing["sha"]
        except RuntimeError as e:
            if "404" not in str(e):
                raise

        encoded_content = base64.b64encode(content.encode("utf-8")).decode("utf-8")

        put_body = {
            "message": commit_message,
            "content": encoded_content,
            "branch": branch,
        }
        if sha:
            put_body["sha"] = sha

        result_data = self._github_request(
            "PUT",
            url,
            github_token,
            json=put_body,
        )

        commit_sha: Optional[str] = None
        if isinstance(result_data, dict):
            commit_sha = (
                result_data.get("commit", {}).get("sha")
                or result_data.get("content", {}).get("sha")
            )

        result = {
            "status": "success",
            "message": "File committed via GitHub API.",
            "file_path": file_path,
            "branch": branch,
            "commit_sha": commit_sha,
        }
        return json.dumps(result, indent=2)

    def _create_pull_request(
        self,
        owner: str,
        repo: str,
        source_branch: str,
        target_branch: str,
        github_token: str,
        title: Optional[str] = None,
        body: Optional[str] = None,
    ) -> str:
        """
        Create a pull request from source_branch -> target_branch.
        """
        if not target_branch:
            raise ValueError("pr_target_branch required.")
        if not github_token:
            raise ValueError("github_token required.")

        url = f"https://github.azc.ext.hp.com/api/v3/repos/{owner}/{repo}/pulls"
        payload = {
            "title": title or f"Auto PR from {source_branch} â†’ {target_branch}",
            "head": source_branch,
            "base": target_branch,
            "body": body or "Automated PR created by GitHub Connector.",
        }
        try:
            pr = self._github_request("POST", url, github_token, json=payload)
            result = {
                "status": "success",
                "message": "Pull request created successfully.",
                "from_branch": source_branch,
                "to_branch": target_branch,
                "pr_number": pr.get("number"),
                "pr_url": pr.get("html_url") or pr.get("url"),
            }
        except RuntimeError as e:
            if "422" in str(e):
                result = {
                    "status": "warning",
                    "message": "A similar pull request may already exist or validation failed.",
                    "from_branch": source_branch,
                    "to_branch": target_branch,
                    "error": str(e),
                }
            else:
                raise
        return json.dumps(result, indent=2)

    # ----------------------------------------------------------------------
    # MAIN ENTRY (used by the platform)
    # ----------------------------------------------------------------------
    def _run(
        self,
        repo_url: str,
        branch: str,
        action: str,
        github_token: Optional[str] = None,
        file_path: Optional[str] = None,
    ) -> Any:
        """
        Main entry point called by the platform.
        """
        self._validate_repo_url(repo_url)
        self._validate_branch(branch)
        self._validate_action(action)

        if action in ("read_file", "commit_file"):
            self._validate_file_path(file_path)

        owner, repo = self._parse_repo(repo_url)

        if action == "list_files":
            return self._list_files(owner, repo, branch, github_token)

        if action == "read_file":
            return self._read_file(owner, repo, branch, file_path, github_token)

        if action == "commit_file":
            if not github_token:
                raise ValueError("github_token required for commit_file.")
            return self._commit_file(
                owner,
                repo,
                branch,
                file_path,
                content,
                commit_message,
                github_token,
            )

        if action == "create_pull_request":
            if not github_token:
                raise ValueError("github_token required for create_pull_request.")
            return self._create_pull_request(
                owner,
                repo,
                source_branch=branch,
                target_branch=pr_target_branch,
                github_token=github_token,
            )
        
        raise ValueError("Unknown action.")
